cmake_minimum_required(VERSION 3.20)
project(adrive_navmesh LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
file(GLOB DETOUR_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/detour/Source/*.cpp")
add_library(adrive_navmesh SHARED bridge.cpp ${DETOUR_SOURCES})
target_include_directories(adrive_navmesh PRIVATE ../../third_party/detour/Include)
target_compile_definitions(adrive_navmesh PRIVATE DT_VIRTUAL_QUERYFILTER=1)
add_executable(navmesh_fixture fixture.cpp ${DETOUR_SOURCES})
target_include_directories(navmesh_fixture PRIVATE ../../third_party/detour/Include)
target_compile_definitions(navmesh_fixture PRIVATE DT_VIRTUAL_QUERYFILTER=1)
if(MSVC)
  target_compile_options(adrive_navmesh PRIVATE /W3 /EHsc)
endif()
